public class OddMagicSquare {
    private int[][] magic;  // 2차원 배열 선언 — private로 은닉
    private int n;           // 마방진 크기 n x n

    public int[][] getMagic() {   // getter — 읽기 전용 공개
        return magic;
    }
    public OddMagicSquare(int n) {   // 생성자 ①
        magic = new int[n][n];     // 2차원 배열 생성+초기화(0으로 자동 채움)
        this.n = n;
    }
    public OddMagicSquare() {        // 생성자 ② — 오버로딩
        this(3);    // this(n) → 생성자①을 재사용(생성자 위임)
    }
    public void make() {   // 시계 방향 대각선 채우기 알고리즘
        int x = 0, y = n / 2;
        for (int i = 1; i <= n*n; i++) {
            int temX = x, temY = y;
            magic[x][y] = i;
            if (x-1 < 0) x = n-1; else x--;   // 위로 이동(벽이면 순환)
            if (y-1 < 0) y = n-1; else y--;   // 왼쪽 이동(벽이면 순환)
            if (magic[x][y] != 0) { x = temX+1; y = temY; }  // 이미 채워짐 → 아래로
        }
    }
    public void print() { /* n x n 행렬 + isMagic() 결과 출력 */ }
    private int sumRow(int row) { /* row행 합계 */ return 0; }
    private int sumCol(int col) { /* col열 합계 */ return 0; }
    private int sumDia() { /* 역대각선 합 */ return 0; }
    private int sumReverseDia() { /* 대각선 합 */ return 0; }
    private boolean isMagic() { /* 모든 행·열·대각선 합이 같은지 검증 */ return true; }
}

클래스 + __init__ + 기본값 인자

class OddMagicSquare:
    # Java의 OddMagicSquare(int n) / OddMagicSquare() 두 생성자를
    # 기본값 매개변수 하나로 통합 (오버로딩이 없는 Python의 관용법)
    def __init__(self, n=3):
        self._magic = [[0] * n for _ in range(n)]  # 2차원 리스트
        self._n = n                                    # _ 접두사 = "private" 컨벤션

    @property
    def magic(self):        # getMagic() 대응 — @property로 getter화
        return self._magic

    def make(self):
        x, y = 0, self._n // 2
        for i in range(1, self._n * self._n + 1):
            tem_x, tem_y = x, y
            self._magic[x][y] = i
            x = self._n - 1 if x - 1 < 0 else x - 1
            y = self._n - 1 if y - 1 < 0 else y - 1
            if self._magic[x][y] != 0:
                x, y = tem_x + 1, tem_y

    def print(self):
        print()
        print(f"{self._n} is magic : {self._is_magic()}")
        for row in self._magic:
            print("\t".join(map(str, row)))

    def _sum_row(self, row):  return sum(self._magic[row])
    def _sum_col(self, col):  return sum(r[col] for r in self._magic)
    def _sum_dia(self):       return sum(self._magic[i][self._n-1-i] for i in range(self._n))
    def _sum_rdia(self):      return sum(self._magic[i][i] for i in range(self._n))

    def _is_magic(self):
        totals = [self._sum_row(i) for i in range(self._n)]
        totals += [self._sum_col(i) for i in range(self._n)]
        totals += [self._sum_rdia(), self._sum_dia()]
        return len(set(totals)) == 1 and totals[0] != 0

public class FourMagicSquare {
    private int[][] magic;
    private int n;
    // ↓↓↓ getMagic(), print(), sumRow(), sumCol(), sumDia(),
    //     sumReverseDia(), isMagic() — s115와 글자 하나 안 틀리고 동일 ↓↓↓

    public FourMagicSquare(int n) { magic = new int[n][n]; this.n = n; }
    public FourMagicSquare() { this(4); }   // 기본 크기만 3→4로 다름

    public void make() {
        makeA();   // ① 1~n²을 순서대로 채움
        makeR();   // ② 대칭 영역 값을 n²+1-값 으로 뒤집음(LUX 패턴)
    }
    private void makeA() {
        for (int i = 0; i < n*n; i++) magic[i/n][i%n] = i+1;
    }
    private void makeR() {
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++) {
                boolean edgeRow = (i < n/4) || (i >= n/4*3);
                boolean midCol  = (j >= n/4 && j < n/4*3);
                boolean edgeCol = (j < n/4) || (j >= n/4*3);
                if ((edgeRow && midCol) || (!edgeRow && edgeCol))
                    magic[i][j] = n*n - (i*n + j);
            }
    }
}

class FourMagicSquare:
    # _magic, _n, print, _sum_row.. 등은 OddMagicSquare와 완전히 동일 코드 (중복)
    def __init__(self, n=4):                    # 기본값만 4로 다름
        self._magic = [[0] * n for _ in range(n)]
        self._n = n

    @property
    def magic(self):
        return self._magic

    def make(self):
        self._make_a()
        self._make_r()

    def _make_a(self):
        for i in range(self._n * self._n):
            self._magic[i // self._n][i % self._n] = i + 1

    def _make_r(self):
        n = self._n
        for i in range(n):
            for j in range(n):
                edge_row = i < n//4 or i >= n//4*3
                mid_col  = n//4 <= j < n//4*3
                edge_col = j < n//4 or j >= n//4*3
                if (edge_row and mid_col) or (not edge_row and edge_col):
                    self._magic[i][j] = n*n - (i*n + j)

    # print(), _sum_row(), _sum_col(), _sum_dia(), _sum_rdia(), _is_magic()
    # → OddMagicSquare와 100% 동일 (생략, 그대로 복사됨)

public class SixMagicSquare {
    private int[][] magic;
    private int n;
    // (생성자, getMagic, print, sumRow.. 는 이전과 동일 — 여전히 중복)

    public void make() {
        makeA(); makeB(); makeCD(); makeMulti(); makeOdd();
    }
    private void makeOdd() {
        // ★ 클래스 간 의존(사용 관계) — Six는 Odd가 있어야 완성된다
        // "6 마방진은 3 마방진을 필요로 한다" — 생성 의존(creates & uses)
        OddMagicSquare odd = new OddMagicSquare(n/2);  // 지역변수로 임시 생성
        odd.make();                              // 3마방진을 완성시키고
        int[][] mm = odd.getMagic();               // 결과만 꺼내 쓴다 (필드로 저장 X)
        for (int i = 0; i < n/2; i++)
            for (int j = 0; j < n/2; j++) {
                magic[i][j]           += mm[i][j];  // 4분면 각각에 동일하게 가산
                magic[i][j+n/2]      += mm[i][j];
                magic[i+n/2][j]      += mm[i][j];
                magic[i+n/2][j+n/2] += mm[i][j];
            }

    }

class MagicException(Exception):        # 커스텀 예외 — Exception 상속
    def __init__(self, message="이런마 방진 생성불가"):
        super().__init__(f"{message}형태의 마방진 생성 불가")


class FactoryMagic:
    _instance = None                    # ① 싱글턴: 클래스 변수에 유일 인스턴스 저장

    def __new__(cls):                  # ① __new__를 오버라이딩해 생성 자체를 통제
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def get_magic_square(self, n):        # ② 팩토리 메서드
        if n <= 2:
            raise MagicException("2보다 작은 수의 마장진은 ")
        if n % 2 != 0:
            return OddMagicSquare(n)
        elif n % 4 == 0:
            return FourMagicSquare(n)
        else:
            return SixMagicSquare(n)

    def __enter__(self):    # ③ with 블록 시작 시 호출 (AutoCloseable의 진입부)
        return self

    def __exit__(self, exc_type, exc_val, exc_tb):  # ③ 블록 종료 시 자동 호출 = close()
        print("FactoryMagic End !!")
        return False       # False = 예외를 삼키지 않고 그대로 전파

🔒 Singleton Pattern
애플리케이션 전체에서 단 하나의 인스턴스만 존재하도록 보장하는 생성 패턴입니다.

구현 3요소
1) 생성자를 private(Java) / __new__ 오버라이딩(Python)으로 외부 차단
2) 자기 자신을 참조하는 static/class 변수(ins, _instance)
3) 유일한 진입점 static 메서드(getInstence())

이 프로젝트에서 쓴 이유
마방진을 만드는 "공장"은 상태를 갖지 않는 순수 로직이므로 여러 개를 만들 필요가 없고, 하나만 존재해도 충분합니다.
🏭 Factory Method Pattern
객체 생성 코드를 별도 메서드/클래스로 옮겨, "무엇을 만들지 결정하는 로직"과 "만든 객체를 사용하는 로직"을 분리하는 생성 패턴입니다.

이 프로젝트의 구현
getMagicSquare(n) 메서드가 n의 특성(홀수/4배수/6n+2)을 보고 Odd/Four/Six 중 어떤 것을 new할지 스스로 결정합니다.

이점
SquareMain은 OddMagicSquare라는 이름을 몰라도 됩니다 — new 키워드가 SquareMain에서 FactoryMagic 내부로 이동했기 때문입니다(제어의 역전, IoC의 축소판).
