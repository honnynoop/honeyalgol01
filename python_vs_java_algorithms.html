BFS — 너비 우선 탐색

Breadth-First Search — 가까운 노드부터 레벨 단위로 탐색

Time: O(V + E) Space: O(V) Queue 기반
Python
collections.deque
deque의 popleft() 로 O(1) dequeue
from collections import deque

def bfs(graph, start):
    visited = set()
    queue = deque([start])
    visited.add(start)
    order = []

    while queue:
        node = queue.popleft()   # O(1)
        order.append(node)

        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)

    return order

# 최단 거리 BFS
def bfs_dist(graph, start):
    dist = {start: 0}
    queue = deque([start])

    while queue:
        node = queue.popleft()
        for nb in graph[node]:
            if nb not in dist:
                dist[nb] = dist[node] + 1
                queue.append(nb)
    return dist

import java.util.*;

public static List<Integer> bfs(
        List<List<Integer>> graph,
        int start, int n) {

    boolean[] visited = new boolean[n];
    Queue<Integer> queue = new ArrayDeque<>();
    List<Integer> order = new ArrayList<>();

    queue.offer(start);
    visited[start] = true;

    while (!queue.isEmpty()) {
        int node = queue.poll();
        order.add(node);

        for (int nb : graph.get(node)) {
            if (!visited[nb]) {
                visited[nb] = true;
                queue.offer(nb);
            }
        }
    }
    return order;
}

public static int[] bfsDist(

        int start, int n) {

    int[] dist = new int[n];
    Arrays.fill(dist, -1);
    dist[start] = 0;
    Queue<Integer> q = new ArrayDeque<>();
    q.offer(start);
    while (!q.isEmpty()) {
        int u = q.poll();
        for (int v : graph.get(u))
            if (dist[v] == -1) {
                dist[v] = dist[u] + 1;
                q.offer(v);
            }
    }
    return dist;
}

def dfs_iterative(graph, start):
    visited = set()
    stack = [start]
    order = []

    while stack:
        node = stack.pop()
        if node in visited:
            continue
        visited.add(node)
        order.append(node)

        for nb in reversed(graph[node]):
            if nb not in visited:
                stack.append(nb)
    return order

# 위상 정렬 (DFS 기반)
def topological_sort(graph, n):
    visited = set()
    result = []
    def dfs(v):
        visited.add(v)
        for nb in graph[v]:
            if nb not in visited:
                dfs(nb)
        result.append(v)
    for v in range(n):
        if v not in visited: dfs(v)
    return result[::-1]

static List<Integer> dfsIterative(
        List<List<Integer>> graph,
        int start, int n) {

    boolean[] vis = new boolean[n];
    Deque<Integer> stack = new ArrayDeque<>();
    List<Integer> order = new ArrayList<>();
    stack.push(start);

    while (!stack.isEmpty()) {
        int v = stack.pop();
        if (vis[v]) continue;
        vis[v] = true;
        order.add(v);
        List<Integer> nb = graph.get(v);
        for (int i = nb.size()-1; i >= 0; i--)
            if (!vis[nb.get(i)])
                stack.push(nb.get(i));
    }
    return order;
}

Flood Fill — 영역 채우기

2D 그리드에서 시작점과 같은 색(값)의 연결된 영역을 새 색으로 채우기 (BFS/DFS)

Time: O(M × N) Space: O(M × N) BFS / DFS / Recursion
Python

from collections import deque

# ── 재귀 DFS 방식 ──
def flood_fill_dfs(grid, r, c, new_color):
    rows, cols = len(grid), len(grid[0])
    old_color = grid[r][c]
    if old_color == new_color:
        return grid

    def dfs(r, c):
        if not (0 <= r < rows and
                0 <= c < cols): return
        if grid[r][c] != old_color: return
        grid[r][c] = new_color
        for dr, dc in [(-1,0),(1,0),
                        (0,-1),(0,1)]:
            dfs(r+dr, c+dc)

    dfs(r, c)
    return grid

# ── BFS 방식 ──
def flood_fill_bfs(grid, r, c, new_color):
    rows, cols = len(grid), len(grid[0])
    old_color = grid[r][c]
    if old_color == new_color:
        return grid

    queue = deque([(r, c)])
    grid[r][c] = new_color

    while queue:
        cr, cc = queue.popleft()
        for dr, dc in [(-1,0),(1,0),
                        (0,-1),(0,1)]:
            nr, nc = cr+dr, cc+dc
            if (0 <= nr < rows and
                0 <= nc < cols and
                grid[nr][nc] == old_color):
                grid[nr][nc] = new_color
                queue.append((nr, nc))
    return grid

import java.util.*;

// ── 재귀 DFS 방식 ──
static void floodFillDFS(
        int[][] grid, int r, int c,
        int oldColor, int newColor) {

    int rows = grid.length, cols = grid[0].length;
    if (r < 0 || r >= rows ||
        c < 0 || c >= cols) return;
    if (grid[r][c] != oldColor) return;

    grid[r][c] = newColor;
    int[] dr = {-1, 1,  0, 0};
    int[] dc = { 0, 0, -1, 1};
    for (int i = 0; i < 4; i++)
        floodFillDFS(grid,
            r+dr[i], c+dc[i], oldColor, newColor);
}

// ── BFS 방식 ──
static void floodFillBFS(
        int[][] grid, int sr, int sc,
        int newColor) {

    int rows = grid.length, cols = grid[0].length;
    int oldColor = grid[sr][sc];
    if (oldColor == newColor) return;

    Queue<int[]> q = new ArrayDeque<>();
    q.offer(new int[]{sr, sc});
    grid[sr][sc] = newColor;

    int[] dr = {-1, 1,  0, 0};
    int[] dc = { 0, 0, -1, 1};

    while (!q.isEmpty()) {
        int[] cur = q.poll();
        for (int i = 0; i < 4; i++) {
            int nr = cur[0]+dr[i];
            int nc = cur[1]+dc[i];
            if (nr >= 0 && nr < rows &&
                nc >= 0 && nc < cols &&
                grid[nr][nc] == oldColor) {
                grid[nr][nc] = newColor;
                q.offer(new int[]{nr, nc});
            }
        }
    }
}

Union Find — 분리 집합

경로 압축 + 랭크 합치기로 거의 O(1)에 집합 연산

Time: O(α(n)) ≈ O(1) Space: O(n) Path Compression + Union by Rank
Python
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.size = [1] * n

    def find(self, x):
        # 경로 압축 (Path Compression)
        if self.parent[x] != x:
            self.parent[x] = self.find(
                self.parent[x])
        return self.parent[x]

    def union(self, x, y):
        # 랭크 기반 합치기 (Union by Rank)
        rx, ry = self.find(x), self.find(y)
        if rx == ry: return False  # 이미 같은 집합
        if self.rank[rx] < self.rank[ry]:
            rx, ry = ry, rx
        self.parent[ry] = rx
        self.size[rx] += self.size[ry]
        if self.rank[rx] == self.rank[ry]:
            self.rank[rx] += 1
        return True

    def connected(self, x, y):
        return self.find(x) == self.find(y)

    def get_size(self, x):
        return self.size[self.find(x)]

# 사용 예시
uf = UnionFind(5)
uf.union(0, 1)
uf.union(1, 2)

print(uf.connected(0, 3))  # False

class UnionFind {
    int[] parent, rank, size;

    UnionFind(int n) {
        parent = new int[n];
        rank   = new int[n];
        size   = new int[n];
        for (int i = 0; i < n; i++) {
            parent[i] = i;
            size[i]   = 1;
        }
    }

    // 경로 압축
    int find(int x) {
        if (parent[x] != x)
            parent[x] = find(parent[x]);
        return parent[x];
    }

    // 랭크 기반 합치기
    boolean union(int x, int y) {
        int rx = find(x), ry = find(y);
        if (rx == ry) return false;
        if (rank[rx] < rank[ry]) {
            int tmp = rx; rx = ry; ry = tmp;
        }
        parent[ry] = rx;
        size[rx] += size[ry];
        if (rank[rx] == rank[ry]) rank[rx]++;
        return true;
    }

    boolean connected(int x, int y) {
        return find(x) == find(y);
    }
    int getSize(int x) {
        return size[find(x)];
    }
}

Kruskal — 최소 신장 트리

간선을 가중치 순 정렬 후 사이클 없이 추가 (Union Find 활용)

Time: O(E log E) Space: O(V + E) Union Find + Sort
Python
def kruskal(n, edges):
    """
    n: 정점 수
    edges: [(weight, u, v), ...]
    """
    edges.sort()        # 가중치 오름차순
    uf = UnionFind(n)
    mst = []
    total_cost = 0

    for w, u, v in edges:
        if uf.union(u, v):  # 사이클 없으면 추가
            mst.append((u, v, w))
            total_cost += w
            if len(mst) == n - 1:
                break   # MST 완성

    return mst, total_cost

# 사용 예시

    (14, 3, 4)
]
mst, cost = kruskal(5, edges)

import java.util.*;

static int kruskal(int n, int[][] edges) {
    // edges[i] = {weight, u, v}
    Arrays.sort(edges, (a, b) -> a[0] - b[0]);

    UnionFind uf = new UnionFind(n);
    int totalCost = 0, edgeCount = 0;
    List<int[]> mst = new ArrayList<>();

    for (int[] e : edges) {
        int w = e[0], u = e[1], v = e[2];
        if (uf.union(u, v)) {
            mst.add(e);
            totalCost += w;
            edgeCount++;
            if (edgeCount == n - 1) break;
        }
    }
    return totalCost;
}

Prim — 최소 신장 트리

현재 MST에서 가장 가까운 정점을 Priority Queue로 확장

Time: O(E log V) Space: O(V + E) Priority Queue + 인접 리스트
Python
import heapq
from collections import defaultdict

def prim(graph, start=0):
    """
    graph[u] = [(weight, v), ...]
    """
    visited = set()
    mst = []
    total = 0

    # (cost, 현재노드, 부모노드)
    heap = [(0, start, -1)]

    while heap:
        cost, u, parent = heapq.heappop(heap)
        if u in visited:
            continue
        visited.add(u)
        total += cost

        if parent != -1:
            mst.append((parent, u, cost))

        for w, v in graph[u]:
            if v not in visited:
                heapq.heappush(heap, (w, v, u))

    return mst, total

static int prim(List<List<int[]>> graph, int n) {
    boolean[] inMST = new boolean[n];
    // {cost, node, parent}
    PriorityQueue<int[]> pq =
        new PriorityQueue<>((a,b) -> a[0]-b[0]);
    pq.offer(new int[]{0, 0, -1});

    int total = 0;
    List<int[]> mst = new ArrayList<>();

    while (!pq.isEmpty()) {
        int[] cur = pq.poll();
        int cost = cur[0], u = cur[1], par = cur[2];

        if (inMST[u]) continue;
        inMST[u] = true;
        total += cost;
        if (par != -1)
            mst.add(new int[]{par, u, cost});

        for (int[] nb : graph.get(u)) {
            int w = nb[0], v = nb[1];
            if (!inMST[v])
                pq.offer(new int[]{w, v, u});
        }
    }
    return total;
}

Dijkstra — 단일 출발 최단 경로

음수 가중치 없는 그래프에서 Priority Queue로 최단 거리 탐색

Time: O(E log V) Space: O(V + E) Priority Queue (Min-Heap)
Python
import heapq
from collections import defaultdict

def dijkstra(graph, start, n):
    """
    graph[u] = [(w, v), ...]  # 인접 리스트
    """
    dist = [float('inf')] * n
    dist[start] = 0
    heap = [(0, start)]   # (거리, 노드)

    while heap:
        d, u = heapq.heappop(heap)

        if d > dist[u]:     # 이미 처리된 노드
            continue

        for w, v in graph[u]:
            new_dist = dist[u] + w
            if new_dist < dist[v]:
                dist[v] = new_dist
                heapq.heappush(heap,
                    (new_dist, v))
    return dist

# 경로 복원 포함 버전
def dijkstra_path(graph, start, end, n):
    dist = [float('inf')] * n
    prev = [-1] * n
    dist[start] = 0
    heap = [(0, start)]

    while heap:
        d, u = heapq.heappop(heap)
        if d > dist[u]: continue
        for w, v in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                prev[v] = u
                heapq.heappush(heap, (dist[v], v))

    # 경로 역추적
    path, node = [], end
    while node != -1:
        path.append(node)
        node = prev[node]
    return dist[end], path[::-1]

import java.util.*;

static int[] dijkstra(
        List<List<int[]>> graph,
        int start, int n) {

    int[] dist = new int[n];
    Arrays.fill(dist, Integer.MAX_VALUE);
    dist[start] = 0;

    // {거리, 노드}
    PriorityQueue<int[]> pq =
        new PriorityQueue<>((a,b) -> a[0]-b[0]);
    pq.offer(new int[]{0, start});

    while (!pq.isEmpty()) {
        int[] cur = pq.poll();
        int d = cur[0], u = cur[1];

        if (d > dist[u]) continue;

        for (int[] nb : graph.get(u)) {
            int w = nb[0], v = nb[1];
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.offer(new int[]{dist[v], v});
            }
        }
    }
    return dist;
}

Floyd-Warshall — 전체 쌍 최단 경로

모든 쌍의 최단 경로를 DP로 계산 — 음수 간선 허용

Time: O(V³) Space: O(V²) DP / 2D 배열
Python
def floyd_warshall(n, edges):
    """
    edges: [(u, v, w), ...]
    반환: dist[i][j] = i→j 최단 거리
    """
    INF = float('inf')
    dist = [[INF] * n for _ in range(n)]

    # 자기 자신 거리 0
    for i in range(n):
        dist[i][i] = 0

    # 직접 연결 간선
    for u, v, w in edges:
        dist[u][v] = min(dist[u][v], w)

    # 핵심: 중간 정점 k를 통한 완화
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]

    # 음수 사이클 감지
    for i in range(n):
        if dist[i][i] < 0:
            raise ValueError("음수 사이클 존재!")

    return dist

# 경로 복원 포함 (next 배열)
def fw_with_path(n, edges):
    dist = [[float('inf')]*n for _ in range(n)]
    nxt  = [[None]*n for _ in range(n)]
    for i in range(n): dist[i][i] = 0
    for u,v,w in edges:
        dist[u][v]=w; nxt[u][v]=v
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k]+dist[k][j] < dist[i][j]:
                    dist[i][j]=dist[i][k]+dist[k][j]
                    nxt[i][j]=nxt[i][k]
    return dist, nxt

static int[][] floydWarshall(
        int n, int[][] edges) {

    int INF = 1_000_000_000;
    int[][] dist = new int[n][n];

    // 초기화
    for (int[] row : dist)
        Arrays.fill(row, INF);
    for (int i = 0; i < n; i++)
        dist[i][i] = 0;
    for (int[] e : edges)
        dist[e[0]][e[1]] =
            Math.min(dist[e[0]][e[1]], e[2]);

    // Floyd-Warshall DP
    for (int k = 0; k < n; k++)
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (dist[i][k] != INF &&
                    dist[k][j] != INF)
                    dist[i][j] = Math.min(
                        dist[i][j],
                        dist[i][k] + dist[k][j]);

    // 음수 사이클 감지
    for (int i = 0; i < n; i++)
        if (dist[i][i] < 0)
            throw new RuntimeException(
                "음수 사이클!");

    return dist;
}
